cmake_minimum_required(VERSION 3.20)
if(NOT DEFINED ENV{PICO_SDK_PATH})
  message(FATAL_ERROR "Set PICO_SDK_PATH to a reviewed, pinned Pico SDK checkout with submodules")
endif()
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(signalkey C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
option(SIGNALKEY_BUILD_CHECK "Link-only check with invalid USB identifiers; never flash" OFF)
if(SIGNALKEY_BUILD_CHECK)
  set(SIGNALKEY_USB_VID 0)
  set(SIGNALKEY_USB_PID 0)
  message(WARNING "BUILD CHECK ONLY: invalid USB identity; no deployable outputs")
elseif(NOT DEFINED SIGNALKEY_USB_VID OR NOT DEFINED SIGNALKEY_USB_PID)
  message(FATAL_ERROR "Supply an authorized development VID and PID. No borrowed shipping identifiers are supplied.")
endif()
pico_sdk_init()
add_executable(signalkey main.c usb_descriptors.c)
set_source_files_properties(main.c usb_descriptors.c PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
target_include_directories(signalkey PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_compile_definitions(signalkey PRIVATE SIGNALKEY_USB_VID=${SIGNALKEY_USB_VID} SIGNALKEY_USB_PID=${SIGNALKEY_USB_PID})
target_link_libraries(signalkey pico_stdlib pico_unique_id hardware_watchdog hardware_pio hardware_clocks tinyusb_device tinyusb_board)
pico_generate_pio_header(signalkey ${CMAKE_CURRENT_LIST_DIR}/ring.pio)
if(NOT SIGNALKEY_BUILD_CHECK)
  foreach(id SIGNALKEY_USB_VID SIGNALKEY_USB_PID)
    if(NOT "${${id}}" MATCHES "^(0[xX][0-9a-fA-F]+|[0-9]+)$")
      message(FATAL_ERROR "${id} must be an integer")
    endif()
    math(EXPR numeric_id "${${id}}")
    if(numeric_id LESS 1 OR numeric_id GREATER 65534)
      message(FATAL_ERROR "${id} must be between 1 and 65534")
    endif()
  endforeach()
  pico_add_extra_outputs(signalkey)
endif()
